titleBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = true
        dragStart = input.Position
        startPos = main.Position
    end
end)

    if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = false
    end
end)

    if dragging and (input.UserInputType == Enum.UserInputType.MouseDelta or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStart
        main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)

local function createTab(name, order)
    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(0.25, 0, 1, 0)
    btn.BackgroundColor3 = Color3.fromRGB(45, 45, 45)
    btn.BorderSizePixel = 0
    btn.Text = name
    btn.TextColor3 = Color3.fromRGB(140, 140, 140)
    btn.Font = Enum.Font.GothamBold
    btn.TextSize = 11
    btn.LayoutOrder = order
    btn.Parent = tabBar

    local frame = Instance.new("ScrollingFrame")
    frame.Size = UDim2.new(1, 0, 1, 0)
    frame.BackgroundTransparency = 1
    frame.BorderSizePixel = 0
    frame.ScrollBarThickness = 3
    frame.ScrollBarImageColor3 = Color3.fromRGB(100, 100, 100)
    frame.CanvasSize = UDim2.new(0, 0, 0, 0)
    frame.AutomaticCanvasSize = Enum.AutomaticSize.Y
    frame.Visible = false
    frame.Parent = content

    local layout = Instance.new("UIListLayout")
    layout.Padding = UDim.new(0, 8)
    layout.SortOrder = Enum.SortOrder.LayoutOrder
    layout.Parent = frame

    local padding = Instance.new("UIPadding")
    padding.PaddingTop = UDim.new(0, 10)
    padding.PaddingLeft = UDim.new(0, 12)
    padding.PaddingRight = UDim.new(0, 12)
    padding.Parent = frame

    tabs[name] = btn
    tabContents[name] = frame

    btn.MouseButton1Click:Connect(function()
        for n, f in pairs(tabContents) do
            f.Visible = false
            tabs[n].TextColor3 = Color3.fromRGB(140, 140, 140)
            tabs[n].BackgroundColor3 = Color3.fromRGB(45, 45, 45)
        end
        frame.Visible = true
        btn.TextColor3 = Color3.fromRGB(220, 220, 220)
        btn.BackgroundColor3 = Color3.fromRGB(65, 65, 65)
    end)

    return btn, frame
end

local function createCheckboxSlider(parent, labelText, min, max, default, onToggle, onSlide)
    local holder = Instance.new("Frame")
    holder.Size = UDim2.new(1, 0, 0, 65)
    holder.BackgroundTransparency = 1
    holder.Parent = parent

    local row = Instance.new("Frame")
    row.Size = UDim2.new(1, 0, 0, 20)
    row.BackgroundTransparency = 1
    row.Parent = holder

    local box = Instance.new("TextButton")
    box.Size = UDim2.new(0, 20, 0, 20)
    box.BackgroundColor3 = Color3.fromRGB(60, 60, 60)
    box.Text = ""
    box.BorderSizePixel = 0
    box.Parent = row

    local boxCorner = Instance.new("UICorner")
    boxCorner.CornerRadius = UDim.new(0, 4)
    boxCorner.Parent = box

    local lbl = Instance.new("TextLabel")
    lbl.Size = UDim2.new(1, -28, 1, 0)
    lbl.Position = UDim2.new(0, 28, 0, 0)
    lbl.BackgroundTransparency = 1
    lbl.Text = labelText .. ": " .. default
    lbl.TextColor3 = Color3.fromRGB(200, 200, 200)
    lbl.Font = Enum.Font.Gotham
    lbl.TextSize = 12
    lbl.TextXAlignment = Enum.TextXAlignment.Left
    lbl.Parent = row

    local track = Instance.new("Frame")
    track.Size = UDim2.new(1, 0, 0, 6)
    track.Position = UDim2.new(0, 0, 0, 32)
    track.BackgroundColor3 = Color3.fromRGB(60, 60, 60)
    track.BorderSizePixel = 0
    track.Parent = holder

    local trackCorner = Instance.new("UICorner")
    trackCorner.CornerRadius = UDim.new(1, 0)
    trackCorner.Parent = track

    local fill = Instance.new("Frame")
    fill.Size = UDim2.new((default - min) / (max - min), 0, 1, 0)
    fill.BackgroundColor3 = Color3.fromRGB(120, 120, 120)
    fill.BorderSizePixel = 0
    fill.Parent = track

    local fillCorner = Instance.new("UICorner")
    fillCorner.CornerRadius = UDim.new(1, 0)
    fillCorner.Parent = fill

    local knob = Instance.new("TextButton")
    knob.Size = UDim2.new(0, 16, 0, 16)
    knob.Position = UDim2.new((default - min) / (max - min), -8, 0.5, -8)
    knob.BackgroundColor3 = Color3.fromRGB(180, 180, 180)
    knob.Text = ""
    knob.BorderSizePixel = 0
    knob.Parent = track

    local knobCorner = Instance.new("UICorner")
    knobCorner.CornerRadius = UDim.new(1, 0)
    knobCorner.Parent = knob

    local enabled = false
    box.MouseButton1Click:Connect(function()
        enabled = not enabled
        box.BackgroundColor3 = enabled and Color3.fromRGB(100, 100, 100) or Color3.fromRGB(60, 60, 60)
        box.Text = enabled and "✓" or ""
        box.TextColor3 = Color3.fromRGB(220, 220, 220)
        box.TextSize = 12
        onToggle(enabled)
    end)

    local sliding = false
    knob.MouseButton1Down:Connect(function() sliding = true end)
    knob.TouchLongPress:Connect(function() sliding = true end)
    uis.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            sliding = false
        end
    end)

    rs.RenderStepped:Connect(function()
        if sliding then
            local mouse = player:GetMouse()
            local rel = math.clamp((mouse.X - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
            local val = math.floor(min + (max - min) * rel)
            fill.Size = UDim2.new(rel, 0, 1, 0)
            knob.Position = UDim2.new(rel, -8, 0.5, -8)
            lbl.Text = labelText .. ": " .. val
            onSlide(val)
        end
    end)
end

local playerTab = tabContents["Player"]
local ws = 16
local jp = 50
local flyEnabled = false
local noclipEnabled = false
local flyConn = nil
local noclipConn = nil

createCheckboxSlider(playerTab, "WalkSpeed", 15, 200, 16,
    function(enabled)
        local char = player.Character
        if char then
            local h = char:FindFirstChild("Humanoid")
            if h then h.WalkSpeed = enabled and ws or 16 end
        end
    end,
    function(val)
        ws = val
        local char = player.Character
        if char then
            local h = char:FindFirstChild("Humanoid")
            if h then h.WalkSpeed = val end
        end
    end
)

createCheckboxSlider(playerTab, "JumpPower", 15, 200, 50,
    function(enabled)
        local char = player.Character
        if char then
            local h = char:FindFirstChild("Humanoid")
            if h then h.JumpPower = enabled and jp or 50 end
        end
    end,
    function(val)
        jp = val
        local char = player.Character
        if char then
            local h = char:FindFirstChild("Humanoid")
            if h then h.JumpPower = val end
        end
    end
)

createCheckbox(playerTab, "Fly", function(enabled)
    flyEnabled = enabled
    if flyConn then flyConn:Disconnect() flyConn = nil end
    local char = player.Character
    if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChild("Humanoid")
    if not hrp or not hum then return end
    if enabled then
        hum.PlatformStand = true
        local bp = Instance.new("BodyPosition")
        bp.MaxForce = Vector3.new(1e5, 1e5, 1e5)
        bp.Position = hrp.Position
        bp.Parent = hrp
        local bg = Instance.new("BodyGyro")
        bg.MaxTorque = Vector3.new(1e5, 1e5, 1e5)
        bg.CFrame = hrp.CFrame
        bg.Parent = hrp
        flyConn = rs.RenderStepped:Connect(function()
            local cam = workspace.CurrentCamera
            local move = Vector3.new(0, 0, 0)
            if uis:IsKeyDown(Enum.KeyCode.W) then move = move + cam.CFrame.LookVector end
            if uis:IsKeyDown(Enum.KeyCode.S) then move = move - cam.CFrame.LookVector end
            if uis:IsKeyDown(Enum.KeyCode.A) then move = move - cam.CFrame.RightVector end
            if uis:IsKeyDown(Enum.KeyCode.D) then move = move + cam.CFrame.RightVector end
            if uis:IsKeyDown(Enum.KeyCode.Space) then move = move + Vector3.new(0, 1, 0) end
            if uis:IsKeyDown(Enum.KeyCode.LeftShift) then move = move - Vector3.new(0, 1, 0) end
            bp.Position = bp.Position + move * 0.5
            bg.CFrame = cam.CFrame
        end)
    else
        hum.PlatformStand = false
        for _, v in pairs(hrp:GetChildren()) do
            if v:IsA("BodyPosition") or v:IsA("BodyGyro") then v:Destroy() end
        end
    end
end)

createCheckbox(playerTab, "NoClip", function(enabled)
    noclipEnabled = enabled
    if noclipConn then noclipConn:Disconnect() noclipConn = nil end
    if enabled then
        noclipConn = rs.Stepped:Connect(function()
            local char = player.Character
            if char then
                for _, p in pairs(char:GetDescendants()) do
                    if p:IsA("BasePart") then p.CanCollide = false end
                end
            end
        end)
    end
end)

player.CharacterAdded:Connect(function(char)
    local h = char:WaitForChild("Humanoid")
    h.WalkSpeed = ws
    h.JumpPower = jp
end)
